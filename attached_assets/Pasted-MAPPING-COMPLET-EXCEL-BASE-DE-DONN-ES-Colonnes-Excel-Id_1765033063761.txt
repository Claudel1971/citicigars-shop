Fonction mapExcelRowToProduct()
javascriptconst mapExcelRowToProduct = (row) => {
  // Normaliser le rabais (enlever le signe - si présent)
  const normaliserRabais = (rabais) => {
    if (!rabais || rabais === 0) return 0;
    return Math.abs(parseFloat(rabais));
  };

  const rabais = normaliserRabais(row['Rabais (%)']);
  const prixUnitaire = parseFloat(row['p. u.']) || 0;
  const qtyPack = parseInt(row['Qte / pack']) || 4;
  const qtyBoite = parseInt(row['Qté/boîte']) || 25;

  // Calculer les prix promo
  const calculerPrixPromo = (prix, rabaisPct) => {
    if (rabaisPct === 0) return null;
    const prixAvecRabais = prix * (1 - rabaisPct / 100);
    return Math.round(prixAvecRabais * 2) / 2; // Arrondi 0.50
  };

  return {
    // Identification
    sku: row['SKU'],
    marque: row['Marque'],
    ligne: row['Ligne / Série'],
    pays: row['Pays'],
    
    // ✅ DIMENSIONS - MAPPING CORRECT
    vitole: row['Vitole'],                      // "Robusto", "Toro"
    format: row['Format'],                      // "Toro", "Churchill"
    dimensions: row['Dimensions (pouces)'],     // "6 × 54" ← CHAMP PRINCIPAL
    dimensionsMM: row['Dimensions (mm)'],       // "152 × 21,4" (à ignorer)
    longueur: parseFloat(row['Lenght']) || null,
    ringGauge: parseInt(row['Ring']) || null,
    
    // Conditionnements
    qteBoite: qtyBoite,
    typePack: qtyPack,
    conditionnements: {
      pack: qtyPack,
      boite: qtyBoite
    },
    
    // Évaluations
    puissance: parseInt(row['Puissance']) || 0,
    rating: parseInt(row['Note']) || null,
    top25: row['Top25'] === 'Oui' || row['Top25'] === 'OUI' || row['Top25'] === true,
    rank: parseInt(row['Rang']) || null,
    year: parseInt(row['Année']) || null,
    
    // Badge CA (Cigar Aficionado)
    badges: {
      rating: row['Note'] && row['Note'] !== 'NA' ? row['Note'] : null
    },
    
    // Prix de base
    prixUnitaire: prixUnitaire,
    
    // Promotions calculées
    promotions: {
      unitaire: {
        actif: rabais > 0,
        pourcentage: rabais,
        prixPromo: calculerPrixPromo(prixUnitaire, rabais)
      },
      pack: {
        actif: rabais > 0,
        pourcentage: rabais,
        prixPromo: calculerPrixPromo(prixUnitaire * qtyPack, rabais)
      },
      boite: {
        actif: rabais > 0,
        pourcentage: rabais,
        prixPromo: calculerPrixPromo(prixUnitaire * qtyBoite, rabais)
      }
    },
    
    // Prix calculés (sans promo)
    prixPack: prixUnitaire * qtyPack,
    prixBoite: prixUnitaire * qtyBoite,
    
    // État par défaut
    inCatalogue: true,
    coupDeCoeur: false,
    stock: 0  // À remplir manuellement après import
  };
};

🔍 Extraction du Ring Gauge (Mise à Jour)
Maintenant simplifié car on a dimensions en pouces :
javascriptconst extractRingGauge = (product) => {
  // PRIORITÉ 1 : Champ ringGauge direct (colonne Ring)
  if (product.ringGauge && !isNaN(product.ringGauge)) {
    return parseInt(product.ringGauge);
  }
  
  // PRIORITÉ 2 : Extraire de dimensions (pouces)
  if (product.dimensions) {
    const match = product.dimensions.match(/(\d+(?:[.,]\d+)?)\s*[×xX]\s*(\d+)/);
    if (match) {
      return parseInt(match[2]); // Le 2ème nombre = ring gauge
    }
  }
  
  // PRIORITÉ 3 : Extraire du vitole si contient dimensions
  if (product.vitole && product.vitole.includes('×')) {
    const match = product.vitole.match(/(\d+)\s*[×xX]\s*(\d+)/);
    if (match) {
      return parseInt(match[2]);
    }
  }
  
  return null;
};

// Catégorisation S/M/L/XL/XXL
const getRingCategory = (product) => {
  const gauge = extractRingGauge(product);
  if (gauge == null) return null;
  
  if (gauge < 42) return "S";
  if (gauge < 48) return "M";
  if (gauge < 52) return "L";
  if (gauge < 56) return "XL";
  return "XXL";
};

✅ Points Critiques

"Dimensions (pouces)" → dimensions (champ principal pour filtre)
"Dimensions (mm)" → dimensionsMM (ignoré par les filtres)
"Ring" → ringGauge (champ direct, prioritaire)
"Rabais (%)" → Toujours normalisé en positif (10, pas -10)
Prix promo → Calculés automatiquement et arrondis à 0.50
Images → Conservées lors du ré-import (associées par SKU)
